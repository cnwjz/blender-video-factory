"""Change Scope Check — verify worktree changes against frozen policy."""

import os

from protocol_guard.frozen.snapshot import verify_frozen_task
from protocol_guard.phase2_min.io_utils import (
    sha256_file, load_task_card, load_frozen_task, load_policy,
    normalize_path, resolve_repo_root, validate_path_in_repo, run_git,

EXCLUDED_BUILTIN = {".git", "reviews/UPLOAD_NEXT"}


def _match_policy(path, entry):
    """Check if a changed path matches a policy entry by path_type."""
    np = normalize_path(path).lower()
    ep = normalize_path(entry["path"]).lower()
    pt = entry.get("path_type", "")
    if pt == "file":
        return np == ep
    elif pt == "directory":
        return np == ep or np.startswith(ep.rstrip("/") + "/") or np.startswith(ep + "/")
    return False


def _is_excluded(path):
    np = normalize_path(path).lower()
    for ex in EXCLUDED_BUILTIN:
        en = normalize_path(ex).lower()
        if np == en or np.startswith(en.rstrip("/") + "/"):
            return True
    return False


def change_scope_check(task_path, frozen_dir):
    """Check worktree changes since baseline comply with frozen policy.

    Returns: (exit_code: int, result: dict)
        0 = PASS, 1 = FAIL (policy violation), 2 = ERROR (invalid input)
    """
    result = {
        "schema_version": "1",
        "checker": "change_scope_check",
        "task_id": None,
        "baseline_commit_sha": None,
        "policy_sha256": None,
        "actual_changes": [],
        "out_of_scope_changes": [],
        "protected_file_changes": [],
        "input_errors": [],
        "result": None,
    }
    errors = result["input_errors"]

    # 1. Verify frozen task
    match, _, _, ferr = verify_frozen_task(task_path, frozen_dir)
    if not match:
        errors.append(f"Frozen task verification failed: {ferr}")
        result["result"] = "ERROR"
        return (2, result)

    # 2. Load task card and frozen task
    frozen_data, ferr2 = load_frozen_task(frozen_dir)
    if frozen_data is None:
        errors.append(ferr2)
        result["result"] = "ERROR"
        return (2, result)

    task_data = load_task_card(task_path)
    task_id = task_data.get("task_id", "unknown")
    result["task_id"] = task_id

    # 3. Read fixed_params
    fixed = frozen_data.get("fixed_params", {})
    if not isinstance(fixed, dict):
        errors.append("fixed_params must be a dict")
        result["result"] = "ERROR"
        return (2, result)

    baseline_commit = fixed.get("baseline_commit_sha")
    policy_path = fixed.get("policy_path")
    policy_sha = fixed.get("policy_sha256")
    worktree_clean = fixed.get("worktree_clean_at_approval")
    result["baseline_commit_sha"] = baseline_commit

    if worktree_clean is not True:
        errors.append(f"worktree_clean_at_approval must be true, got: {worktree_clean!r}")
        result["result"] = "ERROR"
        return (2, result)
    if not baseline_commit or not policy_path or not policy_sha:
        errors.append("Missing required fixed_params")
        result["result"] = "ERROR"
        return (2, result)

    # 4. Validate policy_path syntax
    try:
        normalize_path(policy_path)
    except ValueError as e:
        errors.append(f"policy_path rejected: {e}")
        result["result"] = "ERROR"
        return (2, result)

    # 5. Resolve repo root from task_path
    repo_root, repo_err = resolve_repo_root(task_path)
    if repo_root is None:
        errors.append(repo_err)
        result["result"] = "ERROR"
        return (2, result)

    # 6. Validate task_path and frozen_dir are within repo
    repo_abs = os.path.realpath(repo_root)
    for label, p in [("task_path", task_path), ("frozen_dir", frozen_dir)]:
        pl = os.path.realpath(p)
        if not (pl == repo_abs or pl.startswith(repo_abs + os.sep)):
            errors.append(f"{label} is outside repo")
            result["result"] = "ERROR"
            return (2, result)
    repo_root = repo_abs  # use canonical path for all subsequent file I/O

    # 7. Validate policy_path within repo
    _, pol_err = validate_path_in_repo(policy_path, repo_root, "policy_path")
    if pol_err:
        errors.append(pol_err)
        result["result"] = "ERROR"
        return (2, result)

    # 8. Verify policy SHA256
    policy_abs = os.path.normpath(os.path.join(repo_root, policy_path))
    try:
        actual_policy_sha = sha256_file(policy_abs)
    except Exception as e:
        errors.append(f"Cannot read policy file: {e}")
        result["result"] = "ERROR"
        return (2, result)
    result["policy_sha256"] = policy_sha
    if actual_policy_sha != policy_sha:
        errors.append(f"Policy SHA256 mismatch")
        result["result"] = "ERROR"
        return (2, result)

    # 9. Load and validate policy
    try:
        policy = load_policy(policy_abs)
    except Exception as e:
        errors.append(f"Cannot parse policy: {e}")
        result["result"] = "ERROR"
        return (2, result)
    if policy.get("schema_version") != "1":
        errors.append("Policy schema_version must be '1'")
        result["result"] = "ERROR"
        return (2, result)
    if policy.get("task_id") != task_id:
        errors.append("Policy task_id mismatch")
        result["result"] = "ERROR"
        return (2, result)

    allowed_paths = policy.get("allowed_paths", [])
    denied_paths = policy.get("denied_paths", [])
    protected_files = policy.get("protected_files", [])

    # Validate path_type on all entries
    for label, entries in [("allowed_paths", allowed_paths), ("denied_paths", denied_paths)]:
        for entry in entries:
            pt = entry.get("path_type", "")
            if pt not in ("file", "directory"):
                errors.append(f"{label} entry has invalid path_type: {entry}")
                result["result"] = "ERROR"
                return (2, result)

    # 10. Validate baseline commit
    out, err, rc = run_git(["cat-file", "-e", baseline_commit], repo_root)
    if rc != 0:
        errors.append(f"Baseline commit not resolvable: {baseline_commit}")
        result["result"] = "ERROR"
        return (2, result)

    # 11. Get worktree changes (each git command failure = ERROR)
    git_changes, git_err = _get_changes_or_error(repo_root, baseline_commit)
    if git_err:
        errors.append(git_err)
        result["result"] = "ERROR"
        return (2, result)

    # Build actual_changes list
    all_changed = set()
    for change_type, paths in git_changes.items():
        for p in paths:
            all_changed.add((p, change_type))

    out_of_scope = []
    protected_changes = []

    # 12. Check protected files
    for pf_entry in protected_files:
        pf_path = normalize_path(pf_entry["path"])
        pf_expected_sha = pf_entry.get("sha256", "")
        fp = os.path.join(repo_root, pf_path)

        if not os.path.exists(fp):
            protected_changes.append({
                "path": pf_path, "reason": "missing",
                "expected_sha256": pf_expected_sha, "actual_sha256": None,
            })
            continue
        if os.path.islink(fp):
            protected_changes.append({
                "path": pf_path, "reason": "symlink/junction",
                "expected_sha256": pf_expected_sha, "actual_sha256": None,
            })
            continue
        if not os.path.isfile(fp):
            protected_changes.append({
                "path": pf_path, "reason": "not a regular file",
                "expected_sha256": pf_expected_sha, "actual_sha256": None,
            })
            continue

        actual_sha = sha256_file(fp)
        if actual_sha != pf_expected_sha:
            protected_changes.append({
                "path": pf_path, "reason": "SHA256 mismatch",
                "expected_sha256": pf_expected_sha, "actual_sha256": actual_sha,
            })

    result["protected_file_changes"] = protected_changes

    # 13. Check changed files against policy
    for path, change_type in sorted(all_changed):
        np = normalize_path(path)
        if _is_excluded(np):
            continue
        # Protected files handled separately
        is_protected = any(normalize_path(pf["path"]).lower() == np.lower()
                          for pf in protected_files)
        if is_protected:
            continue

        result["actual_changes"].append({"path": np, "change_type": change_type})

        # Denied check
        if any(_match_policy(path, d) for d in denied_paths):
            out_of_scope.append({"path": np, "change_type": change_type, "reason": "denied_paths match"})
            continue

        # Allowed check
        if not any(_match_policy(path, a) for a in allowed_paths):
            out_of_scope.append({"path": np, "change_type": change_type, "reason": "not in allowed_paths"})

    result["out_of_scope_changes"] = out_of_scope

    # 14. Determine result
    if errors:
        result["result"] = "ERROR"
        return (2, result)
    if out_of_scope or protected_changes:
        result["result"] = "FAIL"
        return (1, result)
    result["result"] = "PASS"
    return (0, result)


def _get_changes_or_error(repo_root, baseline_commit):
    """Get worktree changes. Return (changes_dict, error_string)."""
    from protocol_guard.phase2_min.io_utils import get_worktree_changes as gwc
    changes = {
        "added": [], "modified": [], "deleted": [], "type_changed": [],
        "staged_added": [], "staged_modified": [], "staged_deleted": [],
        "unstaged_modified": [], "unstaged_deleted": [], "untracked": [],
    }

    # Each git command: if failure, return error
    for cmd_args, target_keys in [
        (["diff", "--name-status", baseline_commit, "HEAD"],
         {"A": "added", "M": "modified", "D": "deleted", "T": "type_changed"}),
        (["diff", "--name-status", "--cached"],
         {"A": "staged_added", "M": "staged_modified", "D": "staged_deleted", "T": "type_changed"}),
        (["diff", "--name-status"],
         {"M": "unstaged_modified", "D": "unstaged_deleted", "T": "type_changed"}),
    ]:
        out, err, rc = run_git(cmd_args, repo_root)
        if rc != 0:
            return (None, f"git {' '.join(cmd_args)} failed: {err.strip()}")
        for line in out.strip().split("\n"):
            if not line: continue
            parts = line.split("\t")
            status = parts[0]
            name = parts[-1] if len(parts) > 1 else ""
            if not name: continue
            # Handle renames
            if status.startswith("R") and len(parts) >= 3:
                changes["deleted"].append(parts[1])
                changes["added"].append(parts[2])
            for st_prefix, target in target_keys.items():
                if status.startswith(st_prefix) and name not in changes[target]:
                    changes[target].append(name)

    # Untracked
    out, err, rc = run_git(["ls-files", "--others", "--exclude-standard"], repo_root)
    if rc != 0:
        return (None, f"git ls-files failed: {err.strip()}")
    for line in out.strip().split("\n"):
        if line:
            changes["untracked"].append(line)

    return (changes, None)

"""Shared read-only utilities for Phase 2 Min checkers."""

import hashlib, json, os, subprocess, yaml


def sha256_file(path):
    h = hashlib.sha256()
    with open(path, "rb") as f:
        while True:
            chunk = f.read(8192)
            if not chunk: break
            h.update(chunk)
    return h.hexdigest()


def sha256_bytes(data):
    return hashlib.sha256(data).hexdigest()


def load_task_card(task_path):
    with open(task_path, "r", encoding="utf-8") as f:
        return yaml.safe_load(f)


def load_frozen_task(frozen_dir):
    fp = os.path.join(frozen_dir, "frozen_task.yaml")
    if not os.path.exists(fp):
        return (None, "frozen_task.yaml not found")
    with open(fp, "r", encoding="utf-8") as f:
        return (yaml.safe_load(f), None)


def normalize_path(p):
    """Normalize to forward-slash relative path. Reject dangerous paths."""
    if not isinstance(p, str) or p == "":
        raise ValueError(f"Invalid path: {p!r}")
    if os.path.isabs(p):
        raise ValueError(f"Absolute path rejected: {p}")
    if p.startswith("\\\\") or p.startswith("//"):
        raise ValueError(f"UNC path rejected: {p}")
    if len(p) >= 2 and p[1] == ":" and not p.startswith(p[:2] + "\\"):
        raise ValueError(f"Drive-relative path rejected: {p}")
    segments = p.replace("\\", "/").split("/")
    if ".." in segments:
        raise ValueError(f"Parent traversal rejected: {p}")
    for c in p:
        if ord(c) < 32:
            raise ValueError(f"Control character in path: {p!r}")
    return "/".join(segments)


def paths_equal_ci(a, b):
    """Windows case-insensitive path comparison."""
    return normalize_path(a).lower() == normalize_path(b).lower()


def run_git(args, cwd):
    """Run a read-only git command. Returns (stdout, stderr, returncode)."""
    try:
        r = subprocess.run(
            ["git"] + args, cwd=cwd, capture_output=True, text=True, timeout=10
        )
        return (r.stdout, r.stderr, r.returncode)
    except FileNotFoundError:
        return ("", "git not found", 1)
    except subprocess.TimeoutExpired:
        return ("", "git timeout", 1)


def get_repo_root(cwd):
    out, err, rc = run_git(["rev-parse", "--show-toplevel"], cwd)
    if rc != 0:
        return (None, f"git rev-parse failed: {err.strip()}")
    return (out.strip(), None)


def resolve_repo_root(task_path):
    """Resolve repository root from task_path.

    Runs git rev-parse --show-toplevel from task_path's parent directory.
    Returns (repo_root: str|None, error: str|None).
    """
    task_dir = os.path.dirname(os.path.abspath(task_path))
    repo_root, err = get_repo_root(task_dir)
    if repo_root is None:
        return (None, f"Cannot resolve repo root from task_path: {err}")
    return (repo_root, None)


def validate_path_in_repo(path_str, repo_root, label):
    """Validate that a relative path stays within repo_root.

    Returns (normalized: str|None, error: str|None).
    """
    if not path_str:
        return (None, f"{label} is empty")
    try:
        np = normalize_path(path_str)
    except ValueError as e:
        return (None, f"{label} rejected: {e}")
    abs_path = os.path.normpath(os.path.join(repo_root, np))
    abs_root = os.path.normpath(os.path.abspath(repo_root))
    if not abs_path.startswith(abs_root + os.sep) and abs_path != abs_root:
        return (None, f"{label} resolves outside repo: {path_str}")
    if os.path.islink(abs_path):
        real = os.path.realpath(abs_path)
        if not real.startswith(abs_root + os.sep) and real != abs_root:
            return (None, f"{label} is symlink/junction outside repo: {path_str}")
    return (np, None)


def validate_dir_in_repo(dir_path, repo_root, label):
    """Validate that a directory path exists and stays within repo_root."""
    np, err = validate_path_in_repo(dir_path, repo_root, label)
    if err:
        return (None, err)
    abs_path = os.path.normpath(os.path.join(repo_root, np))
    if not os.path.isdir(abs_path):
        return (None, f"{label} directory not found: {dir_path}")
    return (np, None)


def get_worktree_changes(repo_root, baseline_commit):
    """Get all worktree changes since baseline.

    Returns dict with keys: added, modified, deleted, type_changed, staged_added,
    staged_modified, staged_deleted, unstaged_modified, unstaged_deleted, untracked.
    Each value is a list of normalized paths.
    """
    changes = {
        "added": [], "modified": [], "deleted": [], "type_changed": [],
        "staged_added": [], "staged_modified": [], "staged_deleted": [],
        "unstaged_modified": [], "unstaged_deleted": [],
        "untracked": [],
    }

    def _parse_status(out, prefix_map):
        for line in out.strip().split("\n"):
            if not line: continue
            parts = line.split("\t")
            status = parts[0]
            name = parts[-1] if len(parts) > 1 else ""
            if not name: continue
            if len(status) >= 2 and status[1] == "T":
                changes["type_changed"].append(name)
            for st_prefix, target in prefix_map.items():
                if status.startswith(st_prefix):
                    if name not in changes[target]:
                        changes[target].append(name)

    # Committed changes since baseline
    out, err, rc = run_git(["diff", "--name-status", baseline_commit, "HEAD"], repo_root)
    if rc == 0:
        _parse_status(out, {"A": "added", "M": "modified", "D": "deleted",
                            "R": "added", "T": "type_changed"})
        # Renames: R100 -> old deleted, new added (already handled by A/D of rename parts)
        for line in out.strip().split("\n"):
            if not line: continue
            parts = line.split("\t")
            if parts[0].startswith("R"):
                if len(parts) >= 3:
                    changes["deleted"].append(parts[1])
                    # parts[2] already added via A-like handling

    # Staged changes
    out, err, rc = run_git(["diff", "--name-status", "--cached"], repo_root)
    if rc == 0:
        _parse_status(out, {"A": "staged_added", "M": "staged_modified",
                            "D": "staged_deleted", "T": "type_changed"})

    # Unstaged changes (working tree vs index)
    out, err, rc = run_git(["diff", "--name-status"], repo_root)
    if rc == 0:
        _parse_status(out, {"M": "unstaged_modified", "D": "unstaged_deleted",
                            "T": "type_changed"})

    # Untracked files
    out, err, rc = run_git(["ls-files", "--others", "--exclude-standard"], repo_root)
    if rc == 0:
        for line in out.strip().split("\n"):
            if line:
                changes["untracked"].append(line)

    return changes


def is_path_in_list(path, path_list):
    """Check if normalized path matches any entry in a list of {path} dicts."""
    np = normalize_path(path)
    for entry in path_list:
        ep = normalize_path(entry["path"])
        if np.lower() == ep.lower():
            return True
        # Prefix match for directories
        if ep.endswith("/") and np.lower().startswith(ep.lower()):
            return True
    return False


def find_protected_entry(path, protected_files):
    """Find the protected file entry for a given path (case-insensitive)."""
    np = normalize_path(path)
    for entry in protected_files:
        if normalize_path(entry["path"]).lower() == np.lower():
            return entry
    return None


def find_entry_by_package_path(package_path, entries):
    """Find entry by package_path (case-insensitive)."""
    np = normalize_path(package_path)
    for e in entries:
        if normalize_path(e.get("package_path", "")).lower() == np.lower():
            return e
    return None


def detect_duplicate_package_paths(entries):
    """Detect case-insensitive duplicate package_paths. Returns list of paths."""
    seen = {}
    dups = []
    for e in entries:
        pp = normalize_path(e.get("package_path", "")).lower()
        if pp in seen:
            dups.append(e.get("package_path"))
        seen[pp] = True
    return dups

def upload_package_check(task_path, frozen_dir):
    """Verify upload package integrity against frozen upload spec.

    Returns: (exit_code: int, result: dict)
        0 = PASS, 1 = FAIL, 2 = ERROR
    """

        "schema_version": "1",
        "checker": "upload_package_check",
        "task_id": None,
        "upload_spec_sha256": None,
        "missing_files": [],
        "extra_files": [],
        "hash_mismatches": [],
        "source_issues": [],
        "duplicate_package_paths": [],
        "manifest_errors": [],
        "input_errors": [],
        "result": None,
    }
    errors = result["input_errors"]

    # 1. Verify frozen task
    match, _, _, ferr = verify_frozen_task(task_path, frozen_dir)
    if not match:
        errors.append(f"Frozen task verification failed: {ferr}")
        result["result"] = "ERROR"
        return (2, result)

    task_data = load_task_card(task_path)
    frozen_data, ferr2 = load_frozen_task(frozen_dir)

    upload_spec_sha = fixed.get("upload_spec_sha256")

        return (2, result)

    try:
        spec = load_upload_spec(upload_spec_abs)
    except Exception as e:
        errors.append(f"Cannot parse upload spec: {e}")
        result["result"] = "ERROR"
        return (2, result)

    if spec.get("schema_version") != "1":
        errors.append("Upload spec schema_version must be '1'")
        result["result"] = "ERROR"
        return (2, result)
    if spec.get("task_id") != task_id:
        errors.append(f"Spec task_id mismatch")
        result["result"] = "ERROR"
        return (2, result)

    spec_entries = spec.get("entries", [])
    manifest_filename = spec.get("manifest_filename", "UPLOAD_MANIFEST.json")

    # 8. Validate spec entries paths
    for entry in spec_entries:
        pp = entry.get("package_path", "")
        sp = entry.get("source_path", "")
        if not pp or not sp:
            errors.append("Spec entry missing package_path or source_path")
            result["result"] = "ERROR"
            return (2, result)
        try:
            normalize_path(pp)
            normalize_path(sp)
        except ValueError as e:
            errors.append(f"Spec entry path rejected: {e}")
            result["result"] = "ERROR"
            return (2, result)

    # 9. Detect duplicate package_paths
    dups = detect_duplicate_package_paths(spec_entries)
    if dups:
        result["duplicate_package_paths"] = dups
        result["result"] = "ERROR"
        return (2, result)

    # 10. Validate UPLOAD_NEXT dir exists within repo
    _, ul_err = validate_dir_in_repo("reviews/UPLOAD_NEXT", repo_root, "UPLOAD_NEXT")
    if ul_err:
        errors.append(ul_err)
        result["result"] = "ERROR"
        return (2, result)
    upload_dir = os.path.normpath(os.path.join(repo_root, "reviews", "UPLOAD_NEXT"))

    # 11. Load manifest
    manifest_path = os.path.join(upload_dir, manifest_filename)
    if not os.path.exists(manifest_path):
        errors.append(f"Manifest not found: {manifest_filename}")
        result["result"] = "ERROR"
        return (2, result)
    try:
        manifest = load_manifest(manifest_path)
    except Exception as e:
        result["manifest_errors"].append(f"Cannot parse manifest: {e}")
        result["result"] = "ERROR"
        return (2, result)
    if manifest.get("schema_version") != "1":
        errors.append("Manifest schema_version must be '1'")
        result["result"] = "ERROR"
        return (2, result)
    if manifest.get("task_id") != task_id:
        errors.append(f"Manifest task_id mismatch")
        result["result"] = "ERROR"
        return (2, result)

    manifest_entries = manifest.get("entries", [])

    # 12. Detect duplicate package_paths in manifest
    mdups = detect_duplicate_package_paths(manifest_entries)
    if mdups:
        result["duplicate_package_paths"] = mdups
        result["result"] = "ERROR"
        return (2, result)

    # 13. Detect manifest self-reference
    for me in manifest_entries:
        if normalize_path(me.get("package_path", "")).lower() == normalize_path(manifest_filename).lower():
            result["manifest_errors"].append("Manifest references itself")
            result["result"] = "FAIL"
            return (1, result)

    # 14. Scan actual files in UPLOAD_NEXT
    actual_files = set()
    for fn in os.listdir(upload_dir):
        fp = os.path.join(upload_dir, fn)
        if os.path.islink(fp):
            result["source_issues"].append({"path": fn, "error": "Symlink/junction in UPLOAD_NEXT"})
        elif os.path.isfile(fp):
            actual_files.add(normalize_path(fn))

    # 15. Expected files = manifest entries + manifest itself
    expected_files = {normalize_path(manifest_filename)}
    for entry in spec_entries:
        expected_files.add(normalize_path(entry["package_path"]))

    extra = actual_files - expected_files
    missing = expected_files - actual_files
    if extra:
        result["extra_files"] = sorted(extra)
    if missing:
        result["missing_files"] = sorted(missing)

    # 16. Per-entry validation
    for spec_entry in spec_entries:
        pp = normalize_path(spec_entry["package_path"])
        sp = normalize_path(spec_entry["source_path"])

        m_entry = find_entry_by_package_path(pp, manifest_entries)
        if m_entry is None:
            result["missing_files"].append(pp)
            continue

        m_pp = normalize_path(m_entry.get("package_path", ""))
        m_sp = normalize_path(m_entry.get("source_path", ""))
        m_sha = m_entry.get("sha256", "")

        if m_pp.lower() != pp.lower():
            result["hash_mismatches"].append({"package_path": pp, "error": "Manifest package_path mismatch"})
            continue
        if m_sp.lower() != sp.lower():
            result["source_issues"].append({"package_path": pp, "error": f"source_path mismatch: {m_sp} != {sp}"})
            continue

        pkg_abs = os.path.join(upload_dir, pp)
        if not os.path.exists(pkg_abs) or os.path.islink(pkg_abs):
            result["missing_files"].append(pp)
            continue
        pkg_sha = sha256_file(pkg_abs)
        if pkg_sha != m_sha:
            result["hash_mismatches"].append({"package_path": pp, "error": "Package SHA256 mismatch"})

        # Validate source_path within repo
        _, sp_err = validate_path_in_repo(sp, repo_root, "source_path")
        if sp_err:
            result["source_issues"].append({"package_path": pp, "error": sp_err})
            continue

        src_abs = os.path.normpath(os.path.join(repo_root, sp))
        if not os.path.exists(src_abs) or os.path.islink(src_abs):
            result["source_issues"].append({"package_path": pp, "error": f"Source not found: {sp}"})
            continue
        src_sha = sha256_file(src_abs)
        if src_sha != pkg_sha:
            result["hash_mismatches"].append({"package_path": pp, "error": "Source and package SHA256 differ"})

        if os.path.abspath(src_abs).startswith(os.path.abspath(upload_dir) + os.sep) or os.path.abspath(src_abs) == os.path.abspath(upload_dir):
            result["source_issues"].append({"package_path": pp, "error": "Source inside UPLOAD_NEXT"})

    # 17. Result
    if errors:
        result["result"] = "ERROR"
        return (2, result)
    has_issues = (result["missing_files"] or result["extra_files"] or
                  result["hash_mismatches"] or result["source_issues"] or
                  result["duplicate_package_paths"] or result["manifest_errors"])
    if has_issues:
        result["result"] = "FAIL"
        return (1, result)
    result["result"] = "PASS"
    return (0, result)

# ════════════════════ Unified CS Helper ════════════════════════
def _cs_fixture(td, task_subdir=None, extra_allowed=None, denied=None, protected=None,
                worktree_clean=True, target_changes=None):
    """Unified Change Scope fixture.

    Lifecycle:
      1. Init repo. Create ok.txt. Commit baseline B.
      2. Create policy.json, task.yaml, frozen/ (AFTER B, as allowed artifacts).
      3. Apply target_changes to test the checker against.
      4. Return (repo_root, baseline_sha, task_path, frozen_dir, policy_path).
    """
    repo = _git_init(td)

    # Step 1: business file + baseline B
    _write(repo, "ok.txt", "ok")
    _git_commit(repo, "baseline B — business file only")
    bc = subprocess.run(["git", "rev-parse", "HEAD"], cwd=repo, capture_output=True, text=True).stdout.strip()

    # Step 2: policy
    allowed = [
        {"path": "policy.json", "path_type": "file"},
        {"path": "frozen", "path_type": "directory"},
    ]
    task_rel = "task.yaml"
    if task_subdir:
        task_rel = f"{task_subdir}/task.yaml"
        allowed.append({"path": task_subdir, "path_type": "directory"})
    else:
        allowed.append({"path": "task.yaml", "path_type": "file"})

    if extra_allowed:
        allowed.extend(extra_allowed)

    policy = {"schema_version": "1", "task_id": "TASK_CS",
              "allowed_paths": allowed,
              "denied_paths": denied or [],
              "protected_files": protected or []}
    pp = _write(repo, "policy.json", json.dumps(policy))

    # Step 3: task + frozen
    task = _task_dict("TASK_CS", {
        "baseline_commit_sha": bc,
        "policy_path": "policy.json",
        "policy_sha256": sha256_file(pp),
        "worktree_clean_at_approval": worktree_clean,
    })
    tp = os.path.join(repo, task_rel)
    os.makedirs(os.path.dirname(tp), exist_ok=True)
    with open(tp, "w") as f: yaml.dump(task, f)

    fd = os.path.join(repo, "frozen")
    os.makedirs(fd, exist_ok=True)
    shutil.copy2(tp, os.path.join(fd, "frozen_task.yaml"))
    sha = hashlib.sha256(open(tp, "rb").read()).hexdigest()
    with open(os.path.join(fd, "frozen_task.sha256"), "w") as f: f.write(sha + "\n")

    # Step 4: apply target changes
    if target_changes:
        target_changes(repo)

    return repo, bc, tp, fd, pp
