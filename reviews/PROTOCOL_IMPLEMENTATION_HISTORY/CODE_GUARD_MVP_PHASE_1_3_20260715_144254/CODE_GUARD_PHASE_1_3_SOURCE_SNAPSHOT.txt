def _validate_iso_datetime(value, field_name):
    """Strict datetime validation using fromisoformat. Returns error or None."""
    if not isinstance(value, str) or len(value) == 0:
        return f"{field_name} must be a non-empty ISO 8601 string with timezone"
    try:
        dt = datetime.fromisoformat(value)
    except (ValueError, TypeError):
        return f"{field_name}: invalid ISO 8601 datetime: '{value}'"
    if dt.utcoffset() is None:
        return f"{field_name}: missing timezone (must include Z, +HH:MM, or -HH:MM): '{value}'"
    return None


def _now_iso():
    """Current time as ISO 8601 with timezone."""
    return datetime.now(timezone.utc).astimezone().isoformat()

def validate_patch(actor, patch_fields):
    """Validate that an actor can write the given fields.

    CLAUDE: strict whitelist — only fields in CLAUDE_WRITABLE are allowed.
    Returns (allowed, blocked_fields, reason).
    """
    if not isinstance(patch_fields, dict):
        return (False, [], "patch_fields must be a dict")

    requested = set(patch_fields.keys())

    if actor == "CLAUDE":
        blocked = requested - CLAUDE_WRITABLE
        if blocked:
            return (False, sorted(blocked),
                    f"CLAUDE can only write {sorted(CLAUDE_WRITABLE)}, rejected: {sorted(blocked)}")
        return (True, [], "")

    elif actor == "GPT_PROPOSAL":
        return (False, sorted(requested),
                "GPT_PROPOSAL cannot directly write any field — generate a proposal only")

    elif actor == "USER_APPROVED":
        return (True, [], "")

    else:
        return (False, sorted(requested), f"Unknown actor: {actor}")


def validate_patch_document(patch_doc):
    """Validate a complete patch document against state_patch.schema.json.

    Returns (is_valid, errors).
    """
    import jsonschema
    schema = _load_patch_schema()
    errors = []
    try:
        jsonschema.validate(instance=patch_doc, schema=schema)
    except jsonschema.ValidationError as e:
        errors.append(f"Patch schema violation: {e.message}")
        return False, errors

    # Additional: fields must not contain change_log
    fields = patch_doc.get("fields", {})
    if "change_log" in fields:
        errors.append("change_log cannot be in patch fields — it is auto-generated internally")

    return (len(errors) == 0, errors)


def apply_patch_document(state_data, patch_doc, approval=None):
    """Atomic patch application with full validation chain.

    Steps:
      1. Validate patch document Schema
      2. Validate actor field permissions
      3. Deep-copy state_data
      4. Deep-copy patch fields
      5. Apply fields to candidate state
      6. Auto-generate change_log entry (with GPT_PROPOSAL source tracking)
      7. Validate complete candidate state
      8. Only return success if all steps pass
      9. Any failure → return original state unchanged

    v1.3: GPT_PROPOSAL via USER_APPROVED logs original_actor.
          change_log.reason = "[GPT_PROPOSAL via USER_APPROVED] <reason>"
          actor = "USER_APPROVED"
          Direct USER_APPROVED has no GPT_PROPOSAL prefix.

    approval: required if actor is GPT_PROPOSAL.
    Returns (success, new_state_or_original, errors).
    """
    errors = []

    # Step 1: validate patch document
    ok, errs = validate_patch_document(patch_doc)
    errors.extend(errs)
    if not ok:
        return (False, state_data, errors)

    # Step 2: validate actor field permissions
    original_actor = patch_doc["actor"]
    fields = patch_doc["fields"]
    task_id = patch_doc["task_id"]
    reason = patch_doc.get("reason", "")

    effective_actor = original_actor
    if original_actor == "GPT_PROPOSAL":
        if approval is None or approval.get("approved_by") != "USER_APPROVED":
            errors.append("GPT_PROPOSAL patches require USER_APPROVED approval")
            return (False, state_data, errors)
        effective_actor = "USER_APPROVED"
        approved_fields = approval.get("approved_fields", list(fields.keys()))
        fields = {k: v for k, v in fields.items() if k in approved_fields}

    allowed, blocked, reason_blocked = validate_patch(effective_actor, fields)
    if not allowed:
        errors.append(reason_blocked)
        return (False, state_data, errors)

    # Step 3-4: deep-copy
    candidate = copy.deepcopy(state_data)
    applied_fields = copy.deepcopy(fields)

    # Step 5: apply fields
    candidate.update(applied_fields)

    # Step 6: auto-generate change_log entry
    log_actor = "USER_APPROVED" if original_actor == "GPT_PROPOSAL" else effective_actor
    if original_actor == "GPT_PROPOSAL":
        log_reason = f"[GPT_PROPOSAL via USER_APPROVED] {reason}"
    else:
        log_reason = reason

    log_entry = {
        "timestamp": _now_iso(),
        "actor": log_actor,
        "task_id": task_id,
        "fields_changed": sorted(applied_fields.keys()),
        "reason": log_reason,
    }
    existing_log = list(candidate.get("change_log", []))
    candidate["change_log"] = existing_log + [log_entry]

    # Step 7: validate complete candidate state
    ok_state, state_errs = validate_state(candidate)
    errors.extend(state_errs)
    if not ok_state:
        return (False, state_data, errors)

    # Step 8: success
    return (True, candidate, [])


def apply_patch(state_data, actor, patch_fields, approval=None):
    """Legacy compatibility wrapper around apply_patch_document."""
    patch_doc = {
        "actor": actor,
        "task_id": patch_fields.get("last_task_id", "LEGACY_PATCH"),
        "fields": {k: v for k, v in patch_fields.items() if k != "change_log"},
        "reason": "Legacy apply_patch call",
    }
    if "change_log" in patch_fields:
        return (False, state_data, ["change_log must not be in external patch fields"])
    return apply_patch_document(state_data, patch_doc, approval)

def save_state(state_data, state_path):
    """Atomically save PROJECT_STATE data with full validation.

    Flow:
      1. Deep-copy state_data
      2. validate_state() on the copy
      3. On failure: return (False, errors), no file created/modified
      4. Write to temp file in same directory
      5. Re-read temp file
      6. validate_state() on re-read data
      7. On success: os.replace() temp → target (atomic on same filesystem)
      8. On any failure: delete temp file, return (False, errors)
      9. Original file is never modified unless all checks pass

    Returns (success, errors).
    """
    errors = []

    # Step 1: deep-copy
    candidate = copy.deepcopy(state_data)

    # Step 2: validate before writing
    ok, errs = validate_state(candidate)
    if not ok:
        errors.extend(errs)
        return (False, errors)

    target_dir = os.path.dirname(os.path.abspath(state_path))
    target_name = os.path.basename(state_path)

    try:
        # Step 4: write to temp file in same directory (same filesystem for atomic rename)
        fd, tmp_path = tempfile.mkstemp(
            suffix=".yaml",
            prefix=".project_state_tmp_",
            dir=target_dir,
        )
        try:
            os.close(fd)
            _write_yaml_unchecked(candidate, tmp_path)

            # Step 5: re-read
            with open(tmp_path, "r", encoding="utf-8") as f:
                re_read = yaml.safe_load(f)

            # Step 6: re-validate
            ok_rr, errs_rr = validate_state(re_read)
            if not ok_rr:
                errors.extend(errs_rr)
                return (False, errors)

            # Step 7: atomic replace
            os.replace(tmp_path, state_path)
            return (True, [])

        except Exception:
            # Clean up temp file on any error
            if os.path.exists(tmp_path):
                try:
                    os.remove(tmp_path)
                except OSError:
                    pass
            raise

    except Exception as e:
        errors.append(f"save_state failed: {e}")
        return (False, errors)

class TestSaveState:
    def test_save_valid_state_succeeds(self):
        state = _base_state()
        with tempfile.TemporaryDirectory() as td:
            sp = os.path.join(td, "state.yaml")
            ok, errs = save_state(state, sp)
            assert ok, f"save_state should succeed: {errs}"
            assert os.path.exists(sp)

    def test_save_invalid_state_fails(self):
        state = _base_state()
        state["last_technical_result"] = "BOGUS"
        with tempfile.TemporaryDirectory() as td:
            sp = os.path.join(td, "state.yaml")
            ok, errs = save_state(state, sp)
            assert not ok

    def test_invalid_state_does_not_create_file(self):
        state = _base_state()
        state["last_technical_result"] = "BOGUS"
        with tempfile.TemporaryDirectory() as td:
            sp = os.path.join(td, "state.yaml")
            ok, errs = save_state(state, sp)
            assert not ok
            assert not os.path.exists(sp), "No file should be created for invalid state"

    def test_invalid_write_does_not_corrupt_existing_file(self):
        """Invalid write must not overwrite an existing valid file."""
        with tempfile.TemporaryDirectory() as td:
            sp = os.path.join(td, "state.yaml")
            # Write a valid state first
            state = _base_state()
            ok1, _ = save_state(state, sp)
            assert ok1
            assert os.path.exists(sp)
            with open(sp, "r", encoding="utf-8") as f:
                original_content = f.read()

            # Try to overwrite with invalid state
            bad_state = _base_state()
            bad_state["last_technical_result"] = "BOGUS"
            ok2, _ = save_state(bad_state, sp)
            assert not ok2

            # Original file must be intact
            with open(sp, "r", encoding="utf-8") as f:
                current_content = f.read()
            assert current_content == original_content, "Original file was corrupted"

    def test_save_state_does_not_modify_input(self):
        state = _base_state()
        original = copy.deepcopy(state)
        with tempfile.TemporaryDirectory() as td:
            sp = os.path.join(td, "state.yaml")
            save_state(state, sp)
        assert state == original, "save_state must not mutate input state_data"

    def test_saved_state_revalidates_on_reload(self):
        """After save_state, reloading must pass validate_state."""
        state = _base_state()
        with tempfile.TemporaryDirectory() as td:
            sp = os.path.join(td, "state.yaml")
            ok, _ = save_state(state, sp)
            assert ok
            reloaded = load_state(sp)
            ok_v, errs = validate_state(reloaded)
            assert ok_v, f"Reloaded state failed validation: {errs}"

    def test_temp_file_validation_catches_corruption(self):
        """If re-read data differs, save must fail. (Indirect: save_state re-validates.)"""
        state = _base_state()
        with tempfile.TemporaryDirectory() as td:
            sp = os.path.join(td, "state.yaml")
            ok, _ = save_state(state, sp)
            assert ok
            # Reload and verify
            reloaded = load_state(sp)
            assert reloaded["last_task_id"] == state["last_task_id"]
